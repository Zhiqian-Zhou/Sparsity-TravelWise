% latex/preamble.tex
% =============================================================================
% Preamble for the Sparsity-TravelWise applied-ML pipeline paper.
%
% Design philosophy: keep this file lean and fast-compiling on Overleaf's
% free tier (single-pass < 30 s). Heavy / shell-escape packages that
% previously caused compile-timeouts have been removed:
%
%   • minted        → listings (no Pygments / shell-escape needed)
%   • biblatex+biber → natbib + bibtex (Overleaf fall-back was slow anyway)
%   • todonotes      → unused, removed
%   • algorithm+algpseudocode → unused, removed
%   • microtype      → not loaded (visual gain not worth compile cost)
%
% Section files live in ./sections/01_introduction.tex … 12_conclusion.tex
% plus A_appendix.tex. All figures are physical PNGs under ./figures/
% (no symlinks → Overleaf-portable).
% =============================================================================

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[margin=1in]{geometry}

% Allow long teletype identifiers (e.g. \texttt{cause\_group},
% \texttt{prev\_stop\_actual\_delay}) to break across lines. Without this,
% TeX refuses to break inside an underscore-bearing token and emits an
% Overfull \hbox warning, e.g. on the first paragraph of \S9. The
% emergencystretch knob also helps when paragraph-end boxes don't fit.
\setlength{\emergencystretch}{2em}
\hyphenpenalty=1000
\tolerance=2000

% ── Maths ────────────────────────────────────────────────────────────────────
\usepackage{amsmath, amssymb, amsfonts, mathtools}

% ── Figures + sub-figures + floats ───────────────────────────────────────────
\usepackage{graphicx}
\usepackage{subcaption}
\usepackage{float}
\usepackage{caption}
% NOTE: caption v3 deprecated `bold`; use `bf` instead.
\captionsetup[figure]{font=small, labelfont=bf, justification=raggedright,
                       singlelinecheck=false}
\captionsetup[table]{font=small, labelfont=bf, justification=raggedright,
                      singlelinecheck=false}

% ── Tables ────────────────────────────────────────────────────────────────────
\usepackage{booktabs}
\usepackage{array, multirow, makecell, tabularx}
\usepackage{siunitx}
% siunitx v3 default; we use S columns liberally for decimal alignment in
% result tables. detect-weight + detect-family let \bfseries inside `{...}`
% inside an S cell render bold without re-parsing.
\sisetup{detect-weight=true, detect-family=true,
          group-separator={,}, group-minimum-digits=4,
          round-mode=places, round-precision=3}

% ── Code listings (lightweight; no shell-escape) ─────────────────────────────
% We previously used `minted`, which calls Pygments and tripped the
% Overleaf compile timeout. `listings` is pure-LaTeX and fast.
\usepackage{listings}
\usepackage{xcolor}
\definecolor{lstbg}{RGB}{248,250,252}        % near-white code background
\definecolor{lstkw}{RGB}{30,64,175}          % blue keyword
\definecolor{lstcm}{RGB}{100,116,139}        % grey comment
\definecolor{lststr}{RGB}{5,150,105}         % green string
\lstdefinelanguage{Cypher}{
  morekeywords={MATCH,OPTIONAL,WHERE,RETURN,AS,COUNT,COLLECT,
                  DISTINCT,CREATE,NODE,REL,TABLE,PRIMARY,KEY,FROM,TO,
                  COPY,LOAD,WITH,UNWIND,date,duration,INTERVAL},
  sensitive=true,
  morecomment=[l]{//}, morecomment=[s]{/*}{*/},
  morestring=[b]', morestring=[b]"
}
\lstset{
  basicstyle=\ttfamily\footnotesize,
  keywordstyle=\color{lstkw}\bfseries,
  commentstyle=\color{lstcm}\itshape,
  stringstyle=\color{lststr},
  backgroundcolor=\color{lstbg},
  frame=single, framerule=0.4pt, rulecolor=\color{lstcm!30},
  framesep=4pt, xleftmargin=4pt, xrightmargin=4pt,
  breaklines=true, breakatwhitespace=true,
  showstringspaces=false, columns=fullflexible,
  upquote=true, escapeinside={(*}{*)},
  tabsize=2,
}
% Note: an earlier draft of this preamble carried a `literate=` map that
% transliterated UTF-8 punctuation (em/en dashes, smart quotes, ellipsis,
% NBSP, ge/le) to ASCII-safe equivalents. It collided with Cypher's
% `morestring=[b]'` delimiter (the `'` -> `'` rewrite) and produced
% "Improper alphabetic constant" errors at every \begin{lstlisting} site.
% Every lstlisting/verbatim block is now audited ASCII-clean (verified
% byte-for-byte before each commit), so the defensive map is unnecessary.
% Do NOT re-add it without testing against a Cypher block.

% ── Bibliography (natbib + bibtex — ~3× faster than biblatex/biber) ──────────
\usepackage[numbers, sort&compress, square]{natbib}
\bibliographystyle{plainnat}

% ── Hyperlinks (load BEFORE cleveref) ────────────────────────────────────────
\usepackage[colorlinks=true,
            linkcolor=blue!60!black,
            citecolor=blue!60!black,
            urlcolor=blue!60!black]{hyperref}
\usepackage[capitalise, noabbrev]{cleveref}

% ── Lists, headers, paragraph spacing ────────────────────────────────────────
\usepackage{enumitem}
\setlist{nosep}
\usepackage{parskip}
\usepackage{textcomp}
\usepackage{rotating}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small Stop-level disruption prediction across IT/FI/NL}
\fancyhead[R]{\small \thepage}
\renewcommand{\headrulewidth}{0.4pt}

% ── Math operators / shortcuts ───────────────────────────────────────────────
\DeclareMathOperator{\E}{\mathbb{E}}
\DeclareMathOperator*{\argmax}{arg\,max}
\DeclareMathOperator*{\argmin}{arg\,min}
\newcommand{\R}{\mathbb{R}}
\newcommand{\N}{\mathbb{N}}

% ── Graphics path — all figures live under ./figures/ as physical copies ─────
\graphicspath{{./figures/}}
